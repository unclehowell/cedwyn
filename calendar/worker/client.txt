/* Cedwyn Calendar 0.0.1 — browser script (Tab agent, 31 Oct 2026, task 5ec27d7d). Draws the Day, Week, Month, Year, Compare and Decades
   views from /api/events (the calendar's own derived db). Coverage rules (also in the ⓘ): a day counts when a record is dated to that day;
   a month when a record is dated to that month or a day in it; a year when any record is dated within it (day, month, year or circa).
   Ranges and decades are drawn as bands and never count. Known gaps (missing-records register) are hatched and never count. */
(function () {
  "use strict";
  var root = document.getElementById("cal"); if (!root) return;
  var body = document.getElementById("cal-body"), cov = document.getElementById("cal-cov"), titleB = document.getElementById("cal-title");
  var MN = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
  var MS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"], WD = ["M", "T", "W", "T", "F", "S", "S"];
  var VIEWS = ["day", "week", "month", "year", "compare", "decades"];
  var EV = [], D = {}, MO = {}, YR = {}, GD = {}, GM = {}, GY = {}, BANDS = [], SORTED = [], BYID = {}, META = {};
  var st = { v: "month", y: 1988, m: 12, d: 5, n: 8 };
  function esc(s) { return String(s == null ? "" : s).replace(/[&<>"']/g, function (c) { return { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]; }); }
  function p2(n) { return (n < 10 ? "0" : "") + n; }
  function p4(n) { n = String(n); while (n.length < 4) n = "0" + n; return n; }
  function dim(y, m) { var d = new Date(0); d.setUTCFullYear(y, m, 0); return d.getUTCDate(); }
  function dow(y, m, d) { var x = new Date(0); x.setUTCFullYear(y, m - 1, d); return (x.getUTCDay() + 6) % 7; }
  function add(y, m, d, n) { var x = new Date(0); x.setUTCFullYear(y, m - 1, d + n); return { y: x.getUTCFullYear(), m: x.getUTCMonth() + 1, d: x.getUTCDate() }; }
  function dk(y, m, d) { return p4(y) + "-" + p2(m) + "-" + p2(d); }
  function mk(y, m) { return p4(y) + "-" + p2(m); }
  function yLen(y) { return (y % 4 === 0 && y % 100 !== 0) || y % 400 === 0 ? 366 : 365; }
  function push(o, k, e) { (o[k] = o[k] || []).push(e); }
  function parts(s) { var a = s.split("-"); return { y: +a[0], m: +(a[1] || 0), d: +(a[2] || 0) }; }
  function idx(list) {
    EV = list; list.forEach(function (e, i) {
      e.i = i; var a = parts(e.s); e.y = a.y;
      (e.ids || []).forEach(function (id) { if (!BYID[id]) BYID[id] = e; });
      if (e.p === "span" || e.p === "decade") { var b = e.p === "decade" ? { y: a.y + 9, m: 12, d: 31 } : parts(e.e || e.s); e.a0 = dk(a.y, a.m || 1, a.d || 1); e.a1 = dk(b.y, b.m || 12, b.d || dim(b.y, b.m || 12)); BANDS.push(e); return; }
      if (e.p === "day") push(e.g ? GD : D, dk(a.y, a.m, a.d), e);
      else if (e.p === "month" || (e.p === "circa" && a.m)) push(e.g ? GM : MO, mk(a.y, a.m), e);
      else push(e.g ? GY : YR, p4(a.y), e);
      if (!e.g && (e.p === "day" || e.p === "month")) SORTED.push(e);
    });
  }
  function dayN(y, m, d) { return (D[dk(y, m, d)] || []).length; }
  function monthDays(y, m) { var k = 0, n = dim(y, m); for (var d = 1; d <= n; d++) if (D[dk(y, m, d)]) k++; return k; }
  function monthN(y, m) { var n = (MO[mk(y, m)] || []).length, L = dim(y, m); for (var d = 1; d <= L; d++) n += dayN(y, m, d); return n; }
  function monthGap(y, m) { var n = (GM[mk(y, m)] || []).length, L = dim(y, m); for (var d = 1; d <= L; d++) n += (GD[dk(y, m, d)] || []).length; return n; }
  function yearN(y) { var n = (YR[p4(y)] || []).length; for (var m = 1; m <= 12; m++) n += monthN(y, m); return n; }
  function yearGap(y) { var n = (GY[p4(y)] || []).length; for (var m = 1; m <= 12; m++) n += monthGap(y, m); return n; }
  function lvl(n) { return n <= 0 ? 0 : n === 1 ? 1 : n === 2 ? 2 : n <= 4 ? 3 : 4; }
  function pct(a, b) { return b ? Math.round(100 * a / b) : 0; }
  function bandsIn(a0, a1) { return BANDS.filter(function (e) { return !(e.a1 < a0 || e.a0 > a1); }); }
  function chip(e, extra) { return '<button type="button" class="ch' + (e.g ? " gap" : "") + (extra ? " " + extra : "") + '" data-ev="' + e.i + '" title="' + esc(e.t) + '">' + esc(e.t) + "</button>"; }
  function when(e) {
    var a = parts(e.s);
    if (e.p === "day") return a.d + " " + MN[a.m - 1] + " " + a.y;
    if (e.p === "month") return MN[a.m - 1] + " " + a.y + " (month only)";
    if (e.p === "year") return a.y + " (year only)";
    if (e.p === "circa") return "c. " + (a.m ? MN[a.m - 1] + " " : "") + a.y + " (approximate)";
    if (e.p === "decade") return a.y + "s (decade only)";
    var b = parts(e.e || e.s), f = function (x) { return (x.d ? x.d + " " : "") + (x.m ? MS[x.m - 1] + " " : "") + x.y; };
    return f(a) + " – " + f(b) + " (a range)";
  }
  function band(label, list, open) {
    if (!list.length) return "";
    return '<details class="bd"' + (open ? " open" : "") + '><summary><span>' + esc(label) + "</span><b>" + list.length + "</b></summary><div class=\"bl\">" + list.map(function (e) { return chip(e); }).join("") + "</div></details>";
  }
  function coarse(y, m, a0, a1) {   // the records dated more coarsely than this view, drawn as bands above it
    var out = "";
    if (m) out += band("Sometime in " + MN[m - 1] + " " + y, (MO[mk(y, m)] || []).concat(GM[mk(y, m)] || []), false);
    out += band("Sometime in " + y + " (year only or approximate)", (YR[p4(y)] || []).concat(GY[p4(y)] || []), false);
    out += band("Ranges and decades that include this", bandsIn(a0, a1), false);
    return out ? '<div class="bands">' + out + "</div>" : "";
  }
  function nearest(fromKey, dir) {
    var best = null;
    for (var i = 0; i < SORTED.length; i++) { var s = SORTED[i].s; if (dir > 0 ? s > fromKey && (!best || s < best.s) : s < fromKey && (!best || s > best.s)) best = SORTED[i]; }
    return best;
  }
  function jumpers(k0, k1) {
    var a = nearest(k0, -1), b = nearest(k1 + "~", 1), out = "";
    if (a) out += '<button type="button" class="jp" data-go="' + (a.p === "day" ? "day:" + a.s : "month:" + a.s) + '">← ' + esc(when(a).replace(/ \(.*\)/, "")) + "</button>";
    if (b) out += '<button type="button" class="jp" data-go="' + (b.p === "day" ? "day:" + b.s : "month:" + b.s) + '">' + esc(when(b).replace(/ \(.*\)/, "")) + " →</button>";
    return out ? '<div class="jps"><span class="cal-mut">Nearest dated records</span>' + out + "</div>" : "";
  }
  // ---------- views ----------
  function vDay() {
    var y = st.y, m = st.m, d = st.d, k = dk(y, m, d), L = (D[k] || []), G = (GD[k] || []);
    titleB.textContent = d + " " + MS[m - 1] + " " + y;
    cov.innerHTML = L.length ? "<b>" + L.length + "</b> record" + (L.length > 1 ? "s" : "") + " dated to this day" + (G.length ? " · " + G.length + " known gap" : "") : "No record dated to this day" + (G.length ? " · <b>" + G.length + "</b> known gap" : "");
    var h = coarse(y, m, k, k);
    h += L.concat(G).length ? '<div class="dl">' + L.concat(G).map(function (e) { return '<button type="button" class="card' + (e.g ? " gap" : "") + '" data-ev="' + e.i + '"><span class="k">' + esc(e.ids && e.ids.length ? e.ids.join(", ") : e.ty || "") + "</span><span class=\"t\">" + esc(e.t) + "</span><span class=\"s\">" + esc(e.g ? "Known gap: record not yet disclosed" : (e.ty || "")) + "</span></button>"; }).join("") + "</div>" : '<p class="empty">Nothing in the record for this day yet.</p>';
    h += jumpers(k, k);
    return h;
  }
  function weekStart() { var w = dow(st.y, st.m, st.d); return add(st.y, st.m, st.d, -w); }
  function vWeek() {
    var s = weekStart(), e = add(s.y, s.m, s.d, 6), k0 = dk(s.y, s.m, s.d), k1 = dk(e.y, e.m, e.d), have = 0, rows = "";
    titleB.textContent = s.d + " " + MS[s.m - 1] + (s.y !== e.y ? " " + s.y : "") + " – " + e.d + " " + MS[e.m - 1] + " " + e.y;
    for (var i = 0; i < 7; i++) {
      var x = add(s.y, s.m, s.d, i), k = dk(x.y, x.m, x.d), L = (D[k] || []).concat(GD[k] || []); if (D[k]) have++;
      rows += '<div class="wr lv' + lvl((D[k] || []).length) + ((GD[k] || []).length ? " hasgap" : "") + '"><button type="button" class="wd" data-go="day:' + k + '"><span>' + ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"][i] + "</span><b>" + x.d + "</b></button><div class=\"wc\">" + (L.length ? L.map(function (e) { return chip(e); }).join("") : '<span class="cal-mut">—</span>') + "</div></div>";
    }
    cov.innerHTML = "Days with a dated record: <b>" + have + "</b> of 7";
    return coarse(s.y, s.m === e.m ? s.m : 0, k0, k1) + '<div class="wk">' + rows + "</div>" + (have ? "" : jumpers(k0, k1));
  }
  function vMonth() {
    var y = st.y, m = st.m, L = dim(y, m), first = dow(y, m, 1), have = monthDays(y, m), g = monthGap(y, m), cells = "";
    titleB.textContent = MN[m - 1] + " " + y;
    cov.innerHTML = "Days with a dated record: <b>" + have + "</b> of " + L + " (" + pct(have, L) + "%)" + (g ? " · <span class=\"gk\">" + g + " known gap" + (g > 1 ? "s" : "") + "</span>" : "");
    for (var i = 0; i < first; i++) cells += '<div class="mc out"></div>';
    for (var d = 1; d <= L; d++) {
      var k = dk(y, m, d), E = D[k] || [], G = GD[k] || [], all = E.concat(G);
      cells += '<div class="mc lv' + lvl(E.length) + (G.length ? " hasgap" : "") + '" data-go="day:' + k + '"><span class="dn">' + d + "</span>" + all.slice(0, 2).map(function (e) { return chip(e, "sm"); }).join("") + (all.length > 2 ? '<span class="more">+' + (all.length - 2) + "</span>" : "") + "</div>";
    }
    while ((first + L) % 7) { cells += '<div class="mc out"></div>'; L++; }
    var k0 = dk(y, m, 1), k1 = dk(y, m, dim(y, m));
    return coarse(y, m, k0, k1) + '<div class="mh">' + WD.map(function (w) { return "<span>" + w + "</span>"; }).join("") + '</div><div class="mg">' + cells + "</div>" + (have ? "" : jumpers(k0, k1)) + legend();
  }
  function mini(y, m) {
    var L = dim(y, m), first = dow(y, m, 1), c = "", n = monthN(y, m), md = monthDays(y, m), mo = (MO[mk(y, m)] || []).length, g = monthGap(y, m);
    for (var i = 0; i < first; i++) c += "<i></i>";
    for (var d = 1; d <= L; d++) { var k = dk(y, m, d); c += '<i class="lv' + lvl(dayN(y, m, d)) + ((GD[k] || []).length ? " hasgap" : "") + '"></i>'; }
    return '<button type="button" class="mm' + (mo ? " mband" : "") + (g ? " hasgap" : "") + '" data-go="month:' + mk(y, m) + '"><span class="mt">' + MS[m - 1] + "<b>" + (n ? n : "") + "</b></span><span class=\"mgr\">" + c + "</span>" + (mo ? '<span class="mb">+' + mo + " month only</span>" : "") + "</button>";
  }
  function vYear() {
    var y = st.y, days = 0, months = 0, n = yearN(y), g = yearGap(y), h = "";
    titleB.textContent = String(y);
    for (var m = 1; m <= 12; m++) { days += monthDays(y, m); if (monthN(y, m)) months++; h += mini(y, m); }
    cov.innerHTML = "<b>" + n + "</b> record" + (n === 1 ? "" : "s") + " · days with a record <b>" + days + "</b> of " + yLen(y) + " (" + pct(days, yLen(y)) + "%) · months covered <b>" + months + "</b> of 12" + (g ? " · <span class=\"gk\">" + g + " known gap" + (g > 1 ? "s" : "") + "</span>" : "");
    return coarse(y, 0, p4(y) + "-01-01", p4(y) + "-12-31") + '<div class="yg">' + h + "</div>" + legend();
  }
  function vCompare() {
    var y0 = st.y, N = st.n, rows = "", any = 0, mc = 0;
    titleB.textContent = y0 + " – " + (y0 + N - 1);
    for (var y = y0; y < y0 + N; y++) {
      var n = yearN(y), g = yearGap(y), cells = "", mon = 0;
      for (var m = 1; m <= 12; m++) { var k = monthN(y, m); if (k) mon++; cells += '<button type="button" class="cm lv' + lvl(k) + (monthGap(y, m) ? " hasgap" : "") + '" data-go="month:' + mk(y, m) + '" aria-label="' + MS[m - 1] + " " + y + ": " + k + '">' + (k || "") + "</button>"; }
      if (n) any++; mc += mon;
      var yb = (YR[p4(y)] || []).length, sp = bandsIn(p4(y) + "-01-01", p4(y) + "-12-31").length;
      rows += '<div class="cr"><button type="button" class="cy" data-go="year:' + y + '"><b>' + y + "</b><span>" + pct(mon, 12) + "%</span></button><div class=\"cms\">" + cells + "</div><div class=\"cx\">" + (yb ? '<span class="yb" title="records dated to the year only">+' + yb + "</span>" : "") + (sp ? '<span class="sb" title="ranges that include this year">~' + sp + "</span>" : "") + (g ? '<span class="gb" title="known gaps">' + g + "</span>" : "") + "</div></div>";
    }
    cov.innerHTML = "Years with any record: <b>" + any + "</b> of " + N + " · months covered <b>" + mc + "</b> of " + (12 * N) + " (" + pct(mc, 12 * N) + "%)";
    var sel = '<div class="cn"><span class="cal-mut">Years side by side</span>' + [4, 8, 16, 32].map(function (k) { return '<button type="button" class="cnb' + (k === N ? " on" : "") + '" data-n="' + k + '">' + k + "</button>"; }).join("") + "</div>";
    return sel + '<div class="ch-h"><span></span>' + MS.map(function (x) { return "<span>" + x[0] + "</span>"; }).join("") + '<span></span></div><div class="cmp">' + rows + "</div>" + legend(true);
  }
  function yearAny(y) { return yearN(y) > 0; }
  function vDecades() {
    var c0 = Math.floor(st.y / 100) * 100, rows = "", any = 0, n = 0;
    titleB.textContent = c0 + "s";
    var cents = [], minY = EV.length ? EV[0].y : 1667;
    for (var c = Math.floor(minY / 100) * 100; c <= 2000; c += 100) {
      var a = 0, k = 0; for (var y = c; y < c + 100; y++) { var yn = yearN(y); if (yn) a++; k += yn; }
      if (k || c >= 1600) cents.push('<button type="button" class="ce' + (c === c0 ? " on" : "") + '" data-go="decades:' + c + '"><b>' + (c < 1000 ? "c." + c : c + "s") + "</b><span>" + k + " · " + pct(a, 100) + "%</span></button>");
    }
    for (var dcd = c0; dcd < c0 + 100; dcd += 10) {
      var cells = "", da = 0, dn = 0;
      for (var y2 = dcd; y2 < dcd + 10; y2++) { var t = yearN(y2), g = yearGap(y2); if (t) da++; dn += t; cells += '<button type="button" class="dy lv' + lvl(Math.ceil(t / 3)) + (g ? " hasgap" : "") + '" data-go="year:' + y2 + '"><span>' + String(y2).slice(-2) + "</span>" + (t ? "<b>" + t + "</b>" : "") + "</button>"; }
      any += da; n += dn;
      rows += '<div class="dr"><span class="dl2">' + dcd + "s</span>" + cells + '<span class="dp">' + da + "/10</span></div>";
    }
    var a19 = 0; for (var y3 = 1900; y3 <= 1988; y3++) if (yearAny(y3)) a19++;
    var aAll = 0, y4; for (y4 = 1667; y4 <= 2026; y4++) if (yearAny(y4)) aAll++;
    cov.innerHTML = "This century: years with a record <b>" + any + "</b> of 100 · 1900–1988: <b>" + a19 + "</b> of 89 (" + pct(a19, 89) + "%) · 1667–2026: <b>" + aAll + "</b> of 360 (" + pct(aAll, 360) + "%)";
    return '<div class="cents">' + cents.join("") + '</div><div class="dg">' + rows + "</div>" + legend(true);
  }
  function legend(sc) {
    return '<p class="lg"><span class="sw lv0"></span>none<span class="sw lv1"></span>1<span class="sw lv2"></span>2<span class="sw lv3"></span>3–4<span class="sw lv4"></span>5+' + (sc ? " (per cell)" : "") + ' <span class="sw hasgap"></span>known gap</p>';
  }
  // ---------- state, navigation ----------
  function hashOf() { var v = st.v; return "#" + v + "/" + (v === "day" || v === "week" ? dk(st.y, st.m, st.d) : v === "month" ? mk(st.y, st.m) : v === "compare" ? st.y + "-" + st.n : String(st.y)); }
  function fromHash(h) {
    var m = /^#(day|week|month|year|compare|decades)\/(\d{3,4})(?:-(\d\d?))?(?:-(\d\d?))?/.exec(h || ""); if (!m) return false;
    st.v = m[1]; st.y = +m[2];
    if (st.v === "compare") { st.n = [4, 8, 16, 32].indexOf(+m[3]) >= 0 ? +m[3] : st.n; }
    else { st.m = +(m[3] || st.m || 1); st.d = +(m[4] || 1); if (st.m < 1 || st.m > 12) st.m = 1; if (st.d > dim(st.y, st.m)) st.d = 1; }
    return true;
  }
  function draw(push) {
    var f = { day: vDay, week: vWeek, month: vMonth, year: vYear, compare: vCompare, decades: vDecades }[st.v] || vMonth;
    body.innerHTML = f();
    [].forEach.call(root.querySelectorAll(".cal-views [data-v]"), function (b) { b.setAttribute("aria-selected", b.getAttribute("data-v") === st.v ? "true" : "false"); });
    var h = hashOf(); if (location.hash !== h) { if (push) history.pushState(null, "", h); else history.replaceState(null, "", h); }
    document.title = "Cedwyn Calendar · " + titleB.textContent;
  }
  function step(dir) {
    if (st.v === "day") { var a = add(st.y, st.m, st.d, dir); st.y = a.y; st.m = a.m; st.d = a.d; }
    else if (st.v === "week") { var b = add(st.y, st.m, st.d, 7 * dir); st.y = b.y; st.m = b.m; st.d = b.d; }
    else if (st.v === "month") { st.m += dir; if (st.m < 1) { st.m = 12; st.y--; } if (st.m > 12) { st.m = 1; st.y++; } st.d = 1; }
    else if (st.v === "year") st.y += dir;
    else if (st.v === "compare") st.y += dir * st.n;
    else st.y += 100 * dir;
    draw(true);
  }
  function zoom(dir) { var i = VIEWS.indexOf(st.v) + dir; if (i < 0 || i >= VIEWS.length) return; setView(VIEWS[i]); }
  function setView(v) { if (v === "compare" && st.v !== "compare") st.y = Math.max(1000, st.y - Math.floor(st.n / 2) + 1); if (st.v === "compare" && v !== "compare") st.y = st.y + Math.floor(st.n / 2) - 1; st.v = v; draw(true); }
  function go(t) {
    var a = t.split(":"), v = a[0], p = a[1].split("-");
    st.v = v; st.y = +p[0]; if (p[1]) st.m = +p[1]; if (p[2]) st.d = +p[2]; else if (v === "month" || v === "day") st.d = st.d > dim(st.y, st.m) ? 1 : st.d || 1;
    if (v === "day" && !p[2]) st.d = 1;
    if (v === "compare") st.y = Math.max(1000, st.y - Math.floor(st.n / 2) + 1);
    draw(true);
  }
  // ---------- modal ----------
  function modal(title, html) {
    var dl = document.createElement("dialog"); dl.className = "im";
    dl.innerHTML = '<button type="button" class="x" aria-label="Close">×</button><h3>' + esc(title) + "</h3>" + html;
    document.body.appendChild(dl); dl.addEventListener("close", function () { dl.remove(); });
    dl.addEventListener("click", function (e) { if (e.target === dl || (e.target.closest && e.target.closest(".x"))) dl.close(); });
    if (dl.showModal) dl.showModal(); else dl.setAttribute("open", "");
  }
  var PL = { day: "exact day", month: "month only", year: "year only", circa: "approximate (circa)", decade: "decade only", span: "a range of dates" };
  function evModal(e) {
    var ids = (e.ids || []).map(function (id) { var o = BYID[id], u = o && o.ids[0] === id ? o.u : (o ? o.u : "https://cedwyn.bucklerfamily.estate/api/catalogue/" + id); if (id === (e.ids || [])[0]) u = e.u; return '<a href="' + esc(u) + '" target="_blank" rel="noopener">' + esc(id) + "</a>"; }).join(", ");
    var h = '<p class="md"><b>' + esc(when(e)) + "</b></p>";
    h += '<dl class="mdl"><dt>Date precision</dt><dd>' + esc(PL[e.p] || e.p) + "</dd>" + (e.dd ? "<dt>Date as written</dt><dd>" + esc(e.dd) + "</dd>" : "") + (ids ? "<dt>Catalogue</dt><dd>" + ids + "</dd>" : "") + (e.ty ? "<dt>Type</dt><dd>" + esc(e.ty) + "</dd>" : "") + "<dt>Source</dt><dd>" + esc((e.src || []).join(" · ")) + (e.so ? " · " + esc(e.so) : "") + "</dd>" + (e.h ? "<dt>Should be held by</dt><dd>" + esc(e.h) + "</dd>" : "") + "</dl>";
